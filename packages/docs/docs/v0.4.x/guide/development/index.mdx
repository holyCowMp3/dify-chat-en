import { Tab, Tabs } from 'rspress/theme';

# Local Development and Production Build

Install dependencies:

```bash
pnpm install
```

Start development server:

<Tabs>
  <Tab label="React SPA">
```bash
# Start React SPA application, supports sub-package hot reload, no need to build in advance
pnpm dev
```
	</Tab>
  <Tab label="Next.js">
```bash
# Since Next.js does not support sub-package hot reload yet, sub-packages need to be built in advance
pnpm build:pkgs
# Start Next.js application
pnpm dev:next
```
	</Tab>
</Tabs>

Build production version:

<Tabs>
  <Tab label="React SPA">
```bash
pnpm build
```
	</Tab>
  <Tab label="Next.js">
```bash
pnpm build:next
```
	</Tab>
</Tabs>

Preview production version:

<Tabs>
  <Tab label="React SPA">
```bash
pnpm preview
```
	</Tab>
  <Tab label="Next.js">
```bash
pnpm preview:next
```
	</Tab>
</Tabs>
